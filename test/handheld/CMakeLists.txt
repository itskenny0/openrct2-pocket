cmake_minimum_required(VERSION 3.24)
project(openrct2_handheld_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2>=2.0.14)
add_executable(handheld-input-tests ControllerInputTests.cpp ../../src/openrct2-ui/handheld/ControllerInput.cpp)
target_link_libraries(handheld-input-tests PRIVATE PkgConfig::SDL2)
enable_testing()
add_test(NAME handheld-input COMMAND handheld-input-tests)
if(UNIX)
    add_test(NAME handheld-launcher COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/LauncherTests.sh)
endif()
add_executable(handheld-save-tests SaveSlotsTests.cpp)
add_test(NAME handheld-save-preservation COMMAND handheld-save-tests)
add_executable(handheld-file-browsing-tests FileBrowsingTests.cpp)
add_test(NAME handheld-file-browsing COMMAND handheld-file-browsing-tests)
add_executable(handheld-named-save-tests NamedSaveTests.cpp)
add_test(NAME handheld-named-save COMMAND handheld-named-save-tests)

add_executable(handheld-footer-tests FooterLayoutTests.cpp)
target_compile_options(handheld-footer-tests PRIVATE -fno-char8_t)
target_include_directories(handheld-footer-tests PRIVATE ../../src ../../src/thirdparty)
target_link_libraries(handheld-footer-tests PRIVATE PkgConfig::SDL2)
add_test(NAME handheld-footer COMMAND handheld-footer-tests)
add_executable(handheld-list-scroll-tests ListScrollTests.cpp)
add_test(NAME handheld-list-scroll COMMAND handheld-list-scroll-tests)
add_executable(handheld-menu-feel-tests MenuFeelTests.cpp)
add_test(NAME handheld-menu-feel COMMAND handheld-menu-feel-tests)
